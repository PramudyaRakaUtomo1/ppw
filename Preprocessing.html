tfidf_full = TfidfVectorizer(
    min_df=2,
    max_df=0.95
)

X_full_tfidf = tfidf_full.fit_transform(
    df["berita_clean"]
)

nama_fitur_full = tfidf_full.get_feature_names_out()

df_full_tfidf = pd.DataFrame(
    X_full_tfidf.toarray(),
    columns=nama_fitur_full
)

df_full_tfidf["label"] = df["label_num"]

display(df_full_tfidf.head())

df_full_tfidf.to_excel(
    "data_tfidf_full_dataset.xlsx",
    index=False
)
