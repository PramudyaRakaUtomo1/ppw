Crawling Data

Pada tahap ini dilakukan proses pengumpulan (crawling) artikel berita dari portal detik.com, meliputi kategori finance, untuk membentuk dataset yang nantinya akan diolah pada tahap Preprocessing dan Modeling.

Aktivitas utama:

  • Collect Data - Mengambil URL artikel berita dan menyerap isi artikel secara otomatis
  • Save Dataset - Menyimpan hasil crawling ke dalam file CSV

1. Collect Data

Berikut adalah skrip Python lengkap beserta keluaran (output) hasil running proses crawling berita dari detik.com:

▶ 
import requests
from bs4 import BeautifulSoup
import pandas as pd
import time
from urllib.parse import urljoin

BASE_URL = "https://finance.detik.com"
INDEX_URL = "https://finance.detik.com/indeks"

HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/140.0.0.0 Safari/537.36"
    )
}

TARGET = 100

session = requests.Session()
session.headers.update(HEADERS)

artikel_urls = []
artikel_set = set()


# ==========================================
# 1. MENGAMBIL URL ARTIKEL DARI INDEKS
# ==========================================

def ambil_url_artikel(url_halaman):

    print(f"\nMengambil halaman: {url_halaman}")

    response = session.get(
        url_halaman,
        timeout=15
    )

    response.raise_for_status()

    soup = BeautifulSoup(
        response.text,
        "html.parser"
    )

    for a in soup.find_all("a", href=True):

        href = a["href"]

        # Pastikan URL berasal dari finance.detik.com
        if href.startswith("/"):
            href = urljoin(
                BASE_URL,
                href
            )

        # Hanya artikel finance.detik.com
        if "finance.detik.com" not in href:
            continue

        # Artikel biasanya mempunyai /d-
        if "/d-" not in href:
            continue

        # Hindari duplikat
        if href not in artikel_set:

            artikel_set.add(href)
            artikel_urls.append(href)

            print(
                f"[URL {len(artikel_urls)}] {href}"
            )

        if len(artikel_urls) >= TARGET:
            break


# ==========================================
# 2. CRAWLING HALAMAN INDEKS
# ==========================================

page = 1

while len(artikel_urls) < TARGET:

    if page == 1:
        url = INDEX_URL
    else:
        url = f"{INDEX_URL}?page={page}"

    try:

        jumlah_sebelumnya = len(artikel_urls)

        ambil_url_artikel(url)

        # Jika tidak menemukan URL baru
        if len(artikel_urls) == jumlah_sebelumnya:

            print(
                "Tidak ada URL baru ditemukan."
            )

            break

        page += 1

        time.sleep(1)

    except Exception as e:

        print(
            f"Error halaman {page}: {e}"
        )

        page += 1


print("\n================================")
print("TOTAL URL ARTIKEL:", len(artikel_urls))
print("================================")


# ==========================================
# 3. MENGAMBIL DETAIL ARTIKEL
# ==========================================

data = []

for i, url in enumerate(
    artikel_urls[:TARGET],
    start=1
):

    print(
        f"\n[{i}/{TARGET}] Crawling:"
    )

    print(url)

    try:

        response = session.get(
            url,
            timeout=15
        )

        response.raise_for_status()

        soup = BeautifulSoup(
            response.text,
            "html.parser"
        )


        # -----------------------------
        # JUDUL
        # -----------------------------

        title = soup.find("h1")

        if title:
            title = title.get_text(
                " ",
                strip=True
            )
        else:
            title = ""


        # -----------------------------
        # TANGGAL
        # -----------------------------

        tanggal = ""

        # Beberapa struktur detik
        time_tag = soup.find(
            "time"
        )

        if time_tag:

            tanggal = time_tag.get_text(
                " ",
                strip=True
            )

        else:

            # Coba cari class date
            date_tag = soup.find(
                class_=lambda x:
                x and "date" in str(x).lower()
            )

            if date_tag:

                tanggal = date_tag.get_text(
                    " ",
                    strip=True
                )


        # -----------------------------
        # PENULIS
        # -----------------------------

        penulis = ""

        author_tag = soup.find(
            class_=lambda x:
            x and (
                "author" in str(x).lower()
                or "redaktur" in str(x).lower()
            )
        )

        if author_tag:

            penulis = author_tag.get_text(
                " ",
                strip=True
            )


        # -----------------------------
        # ISI ARTIKEL
        # -----------------------------

        paragraphs = []

        # Cari container artikel
        article_body = soup.find(
            "div",
            class_=lambda x:
            x and (
                "detail__body" in str(x)
                or "article" in str(x).lower()
            )
        )

        if article_body:

            for p in article_body.find_all("p"):

                text = p.get_text(
                    " ",
                    strip=True
                )

                if text:

                    paragraphs.append(
                        text
                    )

        # Jika container tidak ditemukan
        if not paragraphs:

            for p in soup.find_all("p"):

                text = p.get_text(
                    " ",
                    strip=True
                )

                if text:

                    paragraphs.append(
                        text
                    )


        isi = " ".join(
            paragraphs
        )


        # -----------------------------
        # SIMPAN
        # -----------------------------

        data.append({

            "id": i,

            "judul": title,

            "tanggal": tanggal,

            "penulis": penulis,

            "url": url,

            "isi": isi

        })


        print(
            "Berhasil:",
            title
        )

        time.sleep(1)


    except Exception as e:

        print(
            "Gagal:",
            e
        )


# ==========================================
# 4. DATAFRAME
# ==========================================

df = pd.DataFrame(data)


# ==========================================
# 5. BATASI 100 DATA
# ==========================================

df = df.head(100)


# ==========================================
# 6. SIMPAN CSV
# ==========================================

df.to_csv(
    "dataset_detik_finance_100.csv",
    index=False,
    encoding="utf-8-sig"
)


print("\n================================")
print("CRAWLING SELESAI")
print("Jumlah data:", len(df))
print("File: dataset_detik_finance_100.csv")
print("================================")


# ==========================================
# 7. TAMPILKAN DATA
# ==========================================

print(
    df[
        [
            "id",
            "judul",
            "tanggal",
            "penulis",
            "url"
        ]
    ].to_string(
        index=False
    )
)
